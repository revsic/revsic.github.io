function bioToggleMore(btn) {
  var section = btn.closest('.bio-section');
  var expand = !section.classList.contains('bio-section--expanded');
  var animate = !window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var hidden = {height: '0px', paddingTop: '0px', paddingBottom: '0px', marginTop: '0px', marginBottom: '0px',
                borderTopWidth: '0px', borderBottomWidth: '0px', opacity: 0};
  var running = [];

  btn.textContent = expand ? 'Less' : 'More';
  if (expand) section.classList.add('bio-section--expanded');

  section.querySelectorAll('.bio-more-item').forEach(function (el) {
    el.getAnimations().forEach(function (a) { a.cancel(); });
    if (!animate) return;
    var shown = {height: el.offsetHeight + 'px'};
    var frames = expand ? [hidden, shown] : [shown, hidden];
    running.push(el.animate(frames, {duration: 300, easing: 'ease-in-out'}).finished);
  });

  if (!expand) {
    Promise.all(running).then(function () {
      section.classList.remove('bio-section--expanded');
    }, function () {});
  }
}
</script>
